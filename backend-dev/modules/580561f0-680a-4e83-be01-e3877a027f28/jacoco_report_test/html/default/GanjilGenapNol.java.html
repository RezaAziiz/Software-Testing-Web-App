public class GanjilGenapNol {
    public String ganjilGenapNol(int bilangan) {
        if (bilangan == 0) {
            return "Nol";
        } else if (bilangan % 2 == 1) {
            return "Ganjil";
        } else {
            return "Genap";
        }
    }
}
