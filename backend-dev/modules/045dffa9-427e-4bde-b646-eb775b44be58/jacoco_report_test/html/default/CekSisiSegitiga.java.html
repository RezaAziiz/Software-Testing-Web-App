    public boolean cekSisiSegitiga(int sisiSatu, int sisiDua, int sisiTiga) {
        if (sisiSatu + sisiDua > sisiTiga) {
            if (sisiSatu + sisiTiga > sisiDua) {
                if (sisiDua + sisiTiga > sisiSatu) {
                    return true;
                }
            }
        }
        return false;
    }
